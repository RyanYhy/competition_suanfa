cmake_minimum_required(VERSION 3.0.2)
project(my_local_planner)
add_compile_options(-std=c++14 -Wall -Wextra)
find_package(catkin REQUIRED COMPONENTS nav_core base_local_planner costmap_2d pluginlib roscpp tf2_ros tf2_geometry_msgs angles)
catkin_package(INCLUDE_DIRS include LIBRARIES my_planner CATKIN_DEPENDS nav_core base_local_planner costmap_2d pluginlib roscpp tf2_ros tf2_geometry_msgs angles)
include_directories(include ${catkin_INCLUDE_DIRS})
add_library(my_planner src/my_planner.cpp)
target_link_libraries(my_planner ${catkin_LIBRARIES})
install(TARGETS my_planner LIBRARY DESTINATION ${CATKIN_PACKAGE_LIB_DESTINATION})
install(FILES my_local_planner_plugin.xml DESTINATION ${CATKIN_PACKAGE_SHARE_DESTINATION})
install(DIRECTORY include/ DESTINATION ${CATKIN_GLOBAL_INCLUDE_DESTINATION})
