cmake_minimum_required(VERSION 3.0.2)
project(smart_community_sim)
find_package(catkin REQUIRED COMPONENTS roscpp geometry_msgs nav_msgs sensor_msgs std_msgs tf)
find_package(gazebo REQUIRED)
catkin_python_setup()
catkin_package(LIBRARIES community_mecanum_drive CATKIN_DEPENDS roscpp geometry_msgs nav_msgs sensor_msgs std_msgs tf)
include_directories(${catkin_INCLUDE_DIRS} ${GAZEBO_INCLUDE_DIRS})
link_directories(${GAZEBO_LIBRARY_DIRS})
add_library(community_mecanum_drive SHARED src/mecanum_drive_plugin.cpp)
target_compile_features(community_mecanum_drive PRIVATE cxx_std_14)
target_link_libraries(community_mecanum_drive ${catkin_LIBRARIES} ${GAZEBO_LIBRARIES})
add_library(community_signal SHARED src/signal_plugin.cpp)
target_compile_features(community_signal PRIVATE cxx_std_14)
target_link_libraries(community_signal ${catkin_LIBRARIES} ${GAZEBO_LIBRARIES})
catkin_install_python(PROGRAMS scripts/keyboard_teleop.py scripts/verify_keyboard_drive.py scripts/patrol.py scripts/vision_node.py scripts/navigation_benchmark.py scripts/compare_navigation.py scripts/start_cartographer_localization.py scripts/save_cartographer.py scripts/align_cartographer_map.py DESTINATION ${CATKIN_PACKAGE_BIN_DESTINATION})
install(TARGETS community_mecanum_drive community_signal LIBRARY DESTINATION ${CATKIN_PACKAGE_LIB_DESTINATION})
install(DIRECTORY launch worlds materials urdf meshes config maps rviz model DESTINATION ${CATKIN_PACKAGE_SHARE_DESTINATION})
install(FILES README.md NAVIGATION.md requirements-vision.txt DESTINATION ${CATKIN_PACKAGE_SHARE_DESTINATION})
install(PROGRAMS scripts/generate_navigation_maps.py DESTINATION ${CATKIN_PACKAGE_SHARE_DESTINATION}/scripts)
